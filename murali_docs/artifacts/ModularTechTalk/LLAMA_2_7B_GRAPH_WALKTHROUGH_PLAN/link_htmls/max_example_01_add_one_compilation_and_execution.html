16from pathlib import Path
17
18import numpy as np
19from max.driver import CPU, Accelerator, Buffer, accelerator_count
20from max.dtype import DType
21from max.engine import InferenceSession
22from max.graph import DeviceRef, Graph, TensorType, ops
23
24if __name__ == "__main__":
25    mojo_kernels = Path(__file__).parent / "kernels"
26
27    rows = 5
28    columns = 10
29    dtype = DType.float32
30    # Place the graph on a GPU, if available. Fall back to CPU if not.
31    device = CPU() if accelerator_count() == 0 else Accelerator()
32    # Configure our simple one-operation graph.
33    graph = Graph(
34        "addition",
35        # The custom Mojo operation is referenced by its string name, and we
36        # need to provide inputs as a list as well as expected output types.
37        forward=lambda x: (
38            ops.custom(
39                name="add_one",
40                device=DeviceRef.from_device(device),
41                values=[x],
42                out_types=[
43                    TensorType(
44                        dtype=x.dtype,
45                        shape=x.tensor.shape,
46                        device=DeviceRef.from_device(device),
47                    )
48                ],
49            )[0].tensor
50        ),
51        input_types=[
52            TensorType(
53                dtype,
54                shape=[rows, columns],
55                device=DeviceRef.from_device(device),
56            ),
57        ],
58        custom_extensions=[mojo_kernels],
59    )
60
61    # Set up an inference session for running the graph.
62    session = InferenceSession(
63        devices=[device],
64    )
65
66    # Compile the graph.
67    compiled = session.compile(graph)
68    model = session.init(compiled)
69
70    # Fill an input matrix with random values.
71    x_values = np.random.uniform(size=(rows, columns)).astype(np.float32)
72
73    # Create a tensor and move it to the device (CPU or GPU).
74    x = Buffer.from_numpy(x_values).to(device)
75
76    # Run inference with the input tensor.
77    result = model.execute(x)[0]
78
79    # Copy values back to the CPU to be read.
80    assert isinstance(result, Buffer)
81    result = result.to(CPU())
82
83    print("Graph result:")
84    print(result.to_numpy())
85    print()
86
87    print("Expected result:")
88    print(x_values + 1)

32@extensibility.register("add_one")
33struct AddOne:
34    @staticmethod
35    def execute[
36        # The kind of device this will be run on: "cpu" or "gpu"
37        target: StaticString,
38    ](
39        output: OutputTensor,
40        x: InputTensor[dtype=output.dtype, rank=output.rank, ...],
41        # the context is needed for some GPU calls
42        ctx: DeviceContext,
43    ) raises:
44        @inline(.always)
45        def elementwise_add_one[
46            width: Int
47        ](idx: Coord) {var} -> SIMD[x.dtype, width]:
48            return x.load[width](idx) + 1
49
50        foreach[target=target](elementwise_add_one, output, ctx)
