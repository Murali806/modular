          Scope and illustration rules
          • SOURCE: the standard MAX model-worker route for meta-llama/Llama-2-7b-hf, using the built-in Graph API and an ordinary real-device session.
          • EXAMPLE: S1/M1 IDs, selected device combinations, thread settings, two weight-name tiles, token values and reference counts. Default CPU and multiple-device settings demonstrate the constructor; they do not certify that a given Llama configuration can execute on those targets.
          • BOUNDARY: animated driver/GPU submission and completion. No model inference, compilation, benchmark or GPU trace was run to produce these illustrations.
          • A native Engine::Context is owned by the session; the underlying MAX context, CPU runtime and device contexts can be shared. The diagrams do not assign private thread pools or physical GPUs to individual sessions.
          • Lesson 02 shows one ungrouped GraphSpec. Its stored-MEF example assumes a compatible matching entry. Graph tracing precedes this store lookup. Cold compile assumes no lower-level compiled-artifact cache hit.
          • Lesson 03 counts the illustrated decoder's setup once. It does not count all auxiliary graphs/models in MAX Serve. Output Buffer placement remains whatever the model returns.
          • Lesson 04 counts only pictured strong native-session references. Other references, outstanding work and shared resources can extend lifetimes.

172      llvm::sys::Process::GetEnv("MODULAR_PROFILE_FILENAME").value_or("");

174  ContextRef maxContext;

178    Init::Options options;

180        getOrCreateCPUDeviceOptions(config, profileFilename);
181    options.withCPUDeviceOptions(rtOptions);
182    options.withForceDisableCrashReporting(!config.isCrashLogEnabled());
183
184    auto contextOr = Init::getOrCreateContext("max-engine", options);
185    if (contextOr.isError())
186      return contextOr.takeError();

188  }

190  auto &maxTelemetryContext =

193  maxTelemetryContext.setAPILanguage(config.getAPILanguage().str());

195  // Source of truth for the closed set of valid device labels. Downstream
196  // consumers (notably the C-API M_HOST vs M_ACCELERATOR predicate in
197  // M_getDeviceType / M_getDeviceTypeFromSpec) treat anything not rejected
198  // here as a valid device, so this guard must stay in sync with the

200  for (const auto &device : config.getDevices()) {

          183def compile_and_init_graphs(
184    session: InferenceSession, specs: Mapping[str, GraphSpec]
185) -> dict[str, CompiledCallable[..., Any]]:
186    """Compiles and initializes each role's graph.
187
188    Roles that share a group trace into one :class:`~max.graph.Module`,
189    compile once, and initialize once with the weights of all of them; each
190    role's callable then holds the same compiled model. Each compiles through
191    :meth:`~max.engine.InferenceSession.compile_reusing_mefs`, so an ungrouped
192    role reuses the session's precompiled or exported MEFs. A group compiles
193    every time, as any module of several graphs does. Inside
194    :func:`compile_roles_separately`, groups are ignored.
195
196    Each callable is called with one argument per input type, positionally,
197    or with the named input tree as its one argument, and returns the graph's
198    outputs in the tree its ``forward`` returned. It keeps no weight registry,
199    so the host weights can be released after initialization. A ModuleV3
200    role's callable instead takes and returns what the module's ``forward``
201    does, adds the signal buffers a multi-device graph takes, and keeps its
202    weight registry.
203
204    Args:
205        session: The session that compiles and initializes the graphs.
206        specs: The graphs to compile, keyed by role.
207
208    Returns:
209        Each role's compiled, initialized graph.
210
211    Raises:
212        ValueError: If two roles in one group share a graph name, or bind
213            different values to one weight name.
214    """
215    separately = _compile_roles_separately.get()
216    groups: dict[str, list[str]] = {}
217    for role, spec in specs.items():
218        # Ungrouped roles compile alone, so each gets its own key.
219        key = (
220            spec.group
221            if spec.group is not None and not separately
222            else f"\0{role}"
223        )
224        groups.setdefault(key, []).append(role)
225
226    callables: dict[str, CompiledCallable[..., Any]] = {}
227    for members in groups.values():
228        names = [specs[role].name for role in members]
229        if len(set(names)) != len(names):
230            raise ValueError(
231                f"roles {members} share a group but not distinct graph names:"
232                f" {names}"
233            )
234        traced_graph: Graph | GraphModule
235        if len(members) == 1:
236            graphs = {members[0]: specs[members[0]].trace()}
237            traced_graph = graphs[members[0]]
238            extensions = None
239        else:
240            traced_graph = GraphModule()
241            graphs = {role: specs[role].trace(traced_graph) for role in members}
242            # A graph carries its own extensions; a module of graphs does not.
243            extensions = [
244                path
245                for role in members
246                for path in specs[role].custom_extensions
247            ]
248        output_trees = {
249            role: graph.output_tree for role, graph in graphs.items()
250        }
251        # Not session.load_all, which drops the compiled model a callable
252        # keeps to export its MEF.
253        compiled = session.compile_reusing_mefs(
254            traced_graph, custom_extensions=extensions
255        )
256
257        weights: dict[str, DLPackArray] = {}
258        for role in members:
259            for name, value in specs[role].weights_registry().items():
260                if weights.setdefault(name, value) is not value:
261                    raise ValueError(
262                        f"weight {name!r} has two different values"
263                    )
264        by_name = session.init_all(compiled, weights_registry=weights)
265        for role in members:
266            if len(by_name) == 1:
267                # The session can hand back a cached MEF compiled from an
268                # identical graph under another name, so don't look it up by
269                # this spec's name.
270                (model,) = by_name.values()
271            else:
272                model = by_name[specs[role].name]
273            spec = specs[role]
274            if isinstance(spec.module, ModuleV3):
275                # The trace's own signature, so the callable takes and
276                # returns what ``forward`` does and adds any signal buffers.
277                staged, registry = spec._staged
278                callables[role] = CompiledCallable.from_engine_model(
279                    staged, compiled, model, registry
280                )
281            else:
282                callables[role] = _callable(
283                    compiled, model, spec.input_types, output_trees[role]
284                )
285                _check_inputs(role, callables[role].in_tree)
286                _check_outputs(role, callables[role].out_tree)
287    return callables

290M::Core::InferenceSession::loadAll(AsyncValueRef<CompiledModels> compiledModels,
291                                   const nb::dict &weightsDict) {
292  auto [sharedWeightsRegistry, sharedManagedTensors] =
293      buildWeightsFromDict(weightsDict);
294
295  // Ensure the compiled models are ready before accessing individual Model*s.
296  // Release the GIL during the await to allow other Python threads to run
297  // (e.g., for logging progress messages during long compilations).
298  {
299    nb::gil_scoped_release release;
300    if (auto err = Engine::Context::await(compiledModels.copy()))
301      throw std::invalid_argument(err.takeError().get());
302  }
303
304  const CompiledModels &compiled = compiledModels.get();
305  size_t numModels = compiled.getNumModels();
306
307  std::vector<ModelHandle> results;
308  results.reserve(numModels);
309  for (auto &&[index, model] : llvm::enumerate(compiled)) {
310    M::AsyncExecutableModel execModel =
311        load(model.get(), sharedWeightsRegistry);
312    ModelHandle handle(compiledModels.copy(), nb::ref<InferenceSession>(this),
313                       std::move(execModel), sharedWeightsRegistry,
314                       sharedManagedTensors, index);
315    // Build input and output device vectors exactly once at model load time.
316    handle.inputDevicesPy = handle.getInputDevices();
317    handle.outputDevicesPy = handle.getOutputDevices();
318    results.push_back(std::move(handle));
319  }
320  return results;
321}
